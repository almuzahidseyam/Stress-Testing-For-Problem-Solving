==========================================================
Tree with n nodes (So, there must be m = n - 1 edges)
==========================================================
    int n = gen_random(2, 10); // Number of nodes
    cout << n << endl;

    // Create a node permutation suffle it so that the 0-index value can be random and conider as root of the tree
    vector<int> node;
    for (int i = 1; i <= n; i++) node.push_back(i);
    shuffle(node.begin(), node.end(), rng);

    // We will iterate from 1-index. Each it we collect node[i] with a random node fron 0 to i-1 index
    // This will confirm sigle path from source to destination.
    for (int i = 1; i < n; i++) {
        int idx = gen_random(0, i - 1);
        // Edge will be between node[idx] -> node[i]
        cout << node[idx] << " " << node[i] << endl;
    } cout << endl;
